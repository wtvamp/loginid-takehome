294	// reserveUnknownKidRetry reports whether an unknown-kid-triggered refetch
295	// may proceed now for THIS specific kid, and if so atomically marks one
296	// as just having started. Keyed per kid, not by one global timestamp
297	// (Ingrid Solano, 02 cold re-check): KeyForKid runs before signature
298	// verification, so an unauthenticated caller can send a token with a
299	// fabricated kid at no cost — a single global gate lets that caller
300	// permanently occupy the one reservation slot by repeating it every
301	// jwksUnknownKidRefetchInterval, pushing a legitimately-rotated real
302	// key's own refetch back to the full calendar-TTL rollover, silently
303	// recreating most of the original gap this mechanism exists to close.
304	// Per-kid closes that: a garbage kid's repeated reservation never blocks
305	// a DIFFERENT (real) kid's own reservation. jwksMaxTrackedUnknownKids
306	// caps how many distinct kids can hold a reservation at once, so a flood
307	// of distinct fabricated kids still can't hammer the issuer past that
308	// bound — rate-limiting is scoped per-kid, not removed.
309	func (c *JWKSCache) reserveUnknownKidRetry(kid string) bool {
310		c.mu.Lock()
311		defer c.mu.Unlock()
312	
313		if c.unknownKidRetryAt == nil {

316		if last, ok := c.unknownKidRetryAt[kid]; ok && c.now().Sub(last) < jwksUnknownKidRefetchInterval {
317			return false
318		}
319	
320		// Sweep expired entries before checking the size cap, so a kid
321		// that's aged out of its own window doesn't count against another
322		// kid's chance to reserve.
323		for k, t := range c.unknownKidRetryAt {
324			if c.now().Sub(t) >= jwksUnknownKidRefetchInterval {
325				delete(c.unknownKidRetryAt, k)
326			}
327		}
328		if len(c.unknownKidRetryAt) >= jwksMaxTrackedUnknownKids {

330			// one rather than let the map grow further. The ordinary
331			// TTL-driven refresh still eventually picks up a real rotated
332			// key regardless of this outcome.
333			return false
