136	+// GrantLimiter enforces the token-issuing endpoint's own failed-grant
137	+// rate limiting and progressive backoff — handoff-03-auth.md v3:
138	+// "Token-endpoint brute-force protection, separate from the resource-API
139	+// rate limits ... failed-grant rate limiting and progressive backoff per
140	+// client_id and per source IP on the token-issuing endpoint itself, plus
141	+// alerting on a sustained run of failed grants against one client_id."
142	+//
143	+// This is genuinely separate from RateLimiter (ratelimit.go): that type
144	+// gates the resource API by an already-authenticated sub, at a fixed
145	+// per-minute rate; this one gates the endpoint that issues the
146	+// credential in the first place, keyed independently by client_id AND
147	+// by source IP — a distributed brute force (many IPs, one client_id)
148	+// and a single-source credential-stuffing run (one IP, many client_ids)
149	+// are both attack shapes this is meant to slow down, and either key
150	+// exceeding its own backoff blocks the attempt.
151	+//
152	+// Not wired to a live endpoint by this story: /auth/token's real grant
153	+// logic is LT-51's scope (this story's non-goals). This is the
154	+// ready-to-use component LT-51 constructs and calls from the real
155	+// handler — building it now, tested in isolation, means LT-51 doesn't
156	+// also have to design a backoff/alerting scheme under its own deadline.
157	+type GrantLimiter interface {
158	+	// Allow reports whether a grant attempt for clientID from sourceIP
159	+	// may proceed right now — false while either key is in backoff.
160	+	Allow(ctx context.Context, clientID, sourceIP string) (bool, error)
161	+	// RecordFailure records a failed grant attempt for both clientID and
162	+	// sourceIP, advancing each key's backoff state independently.
163	+	// Crossing the alert threshold for a given client_id (not source IP
164	+	// — the requirement names client_id specifically) fires alert
165	+	// exactly once per crossing, not once per subsequent failure.
166	+	RecordFailure(ctx context.Context, clientID, sourceIP string)
167	+	// RecordSuccess clears backoff state for both clientID and
168	+	// sourceIP — a successful grant resets the count, standard
169	+	// exponential-backoff practice: this deters sustained brute force,
170	+	// it does not permanently degrade a legitimate client that mistyped
171	+	// a secret once.
172	+	RecordSuccess(ctx context.Context, clientID, sourceIP string)
173	+}

175	+// grantBackoffState is one key's (a client_id's, or a source IP's)
176	+// consecutive-failure bookkeeping.
177	+type grantBackoffState struct {
178	+	consecutiveFailures int
179	+	blockedUntil        time.Time
180	+	alerted             bool
181	+}
182	+
183	+// InProcessGrantLimiter is an in-process implementation. As with
184	+// RateLimiter/TouchCounter's in-process implementations, a shared store
185	+// is the multi-instance replacement behind the same interface — the
186	+// mechanism is this track's call, per handoff-03-auth.md v3.
187	+//
188	+// Both maps are swept periodically (see sweepLocked) rather than left to
189	+// grow forever: this component's own doc comment names "a distributed
190	+// brute force (many IPs, one client_id)" as a defended attack shape, and
191	+// that is exactly the traffic pattern that would otherwise grow
192	+// bySourceIP without bound — every distinct attacking IP earns a
193	+// permanent entry that only a matching RecordSuccess would remove, which
194	+// an attacking IP by definition never triggers (Nolan Reyes, PR #30
195	+// review — the same unbounded-growth shape an earlier PR's TouchCounter
196	+// review already found and fixed once).
197	+type InProcessGrantLimiter struct {
198	+	mu             sync.Mutex
199	+	byClientID     map[string]*grantBackoffState
200	+	bySourceIP     map[string]*grantBackoffState
201	+	baseBackoff    time.Duration
202	+	maxBackoff     time.Duration
203	+	alertThreshold int
204	+	alertFn        func(clientID string, consecutiveFailures int)
205	+	now            func() time.Time
206	+	lastSweep      time.Time
207	+}

209	+// grantSweepInterval and grantEntryTTL bound how long a key's state
210	+// survives with no further failures against it. A blocked key is never
211	+// swept before its own backoff has elapsed (sweeping mid-block would
212	+// silently un-block it early); grantEntryTTL is generous past any
213	+// realistic maxBackoff so a legitimately still-escalating attacker isn't
214	+// reset early either.

220	+// NewInProcessGrantLimiter constructs a limiter with exponential backoff
221	+// starting at baseBackoff (doubling per consecutive failure, capped at
222	+// maxBackoff) and alertFn called once per client_id crossing
223	+// alertThreshold consecutive failures. alertFn may be nil (no alerting —
224	+// tests use this; production wiring passes a real sink).
225	+func NewInProcessGrantLimiter(baseBackoff, maxBackoff time.Duration, alertThreshold int, alertFn func(clientID string, consecutiveFailures int)) *InProcessGrantLimiter {
226	+	return &InProcessGrantLimiter{
227	+		byClientID:     make(map[string]*grantBackoffState),
228	+		bySourceIP:     make(map[string]*grantBackoffState),
229	+		baseBackoff:    baseBackoff,
230	+		maxBackoff:     maxBackoff,

237	+func (l *InProcessGrantLimiter) Allow(_ context.Context, clientID, sourceIP string) (bool, error) {
238	+	l.mu.Lock()
239	+	defer l.mu.Unlock()
240	+	now := l.now()
241	+	l.sweepLocked(now)
242	+	return !isBlocked(l.byClientID[clientID], now) && !isBlocked(l.bySourceIP[sourceIP], now), nil
243	+}
244	+
245	+// sweepLocked drops any entry whose blockedUntil is at least
246	+// grantEntryTTL in the past — called with l.mu already held, at most
247	+// once per grantSweepInterval, so it adds no per-call cost beyond a
248	+// single time comparison on every other call.
249	+func (l *InProcessGrantLimiter) sweepLocked(now time.Time) {
250	+	if now.Sub(l.lastSweep) < grantSweepInterval {
251	+		return
252	+	}
253	+	l.lastSweep = now
254	+	sweepMap(l.byClientID, now)
255	+	sweepMap(l.bySourceIP, now)
256	+}
257	+
258	+func sweepMap(m map[string]*grantBackoffState, now time.Time) {
259	+	for key, s := range m {
260	+		if now.Sub(s.blockedUntil) >= grantEntryTTL {
261	+			delete(m, key)
262	+		}
263	+	}
264	+}
265	+
266	+func isBlocked(s *grantBackoffState, now time.Time) bool {
267	+	return s != nil && now.Before(s.blockedUntil)
268	+}
269	+
270	+func (l *InProcessGrantLimiter) RecordFailure(_ context.Context, clientID, sourceIP string) {
271	+	l.mu.Lock()
272	+	defer l.mu.Unlock()
273	+	now := l.now()
274	+
275	+	cs := l.advance(l.byClientID, clientID, now)
276	+	l.advance(l.bySourceIP, sourceIP, now)
277	+
278	+	if cs.consecutiveFailures >= l.alertThreshold && !cs.alerted {
279	+		cs.alerted = true
280	+		if l.alertFn != nil {
281	+			l.alertFn(clientID, cs.consecutiveFailures)
282	+		}
283	+	}
284	+}
285	+
286	+// advance increments key's consecutive-failure count and sets its
287	+// exponential backoff window, creating the state entry if this is its
288	+// first failure.
289	+func (l *InProcessGrantLimiter) advance(m map[string]*grantBackoffState, key string, now time.Time) *grantBackoffState {
290	+	s, ok := m[key]
291	+	if !ok {
292	+		s = &grantBackoffState{}
293	+		m[key] = s
294	+	}
295	+	s.consecutiveFailures++
296	+	backoff := l.baseBackoff << uint(s.consecutiveFailures-1) // 2^(n-1) * base
297	+	if backoff > l.maxBackoff || backoff <= 0 {               // overflow guard: a left-shifted duration can wrap negative
298	+		backoff = l.maxBackoff
299	+	}
300	+	s.blockedUntil = now.Add(backoff)
301	+	return s
302	+}
303	+
304	+func (l *InProcessGrantLimiter) RecordSuccess(_ context.Context, clientID, sourceIP string) {
305	+	l.mu.Lock()
306	+	defer l.mu.Unlock()
307	+	delete(l.byClientID, clientID)
308	+	delete(l.bySourceIP, sourceIP)
309	+}
