// ValidateProfileQuery enforces the ErrInvalidQuery triggers enumerated in
// multi-db-strategy.md §4: a non-nil filter that is empty after trimming;
// Limit < 0 or > 100; Offset < 0 or > 10_000; Country not exactly two
// characters. An all-nil query is legal. Both backends call this before
// building any SQL, so the rule is identical across engines rather than
// each package re-deriving it.
func ValidateProfileQuery(q ProfileQuery) error {
	for _, f := range []*string{q.Name, q.Phone, q.Region, q.Country} {
		if f != nil && strings.TrimSpace(*f) == "" {
			return ErrInvalidQuery
		}
	}
	if q.Limit < 0 || q.Limit > 100 {
		return ErrInvalidQuery
	}
	if q.Offset < 0 || q.Offset > 10_000 {
		return ErrInvalidQuery
	}
	if q.Country != nil && len(*q.Country) != 2 {
		return ErrInvalidQuery
	}
	return nil
}

// section's SQL — the real deliverable in migrations/{shared,postgres}/,
// not a duplicated copy of the schema. Applying the actual migration files
// in tests is what keeps this test harness honest: if the migrations
// drift from what the Go code expects, the integration tests fail, not
// just a hand-maintained fixture.
func migrationSQL(t *testing.T, path string) string {
	t.Helper()
	b, err := os.ReadFile(path)
	if err != nil {
		t.Fatalf("reading migration %s: %v", path, err)
	}
	content := string(b)
	upStart := strings.Index(content, "-- +goose Up")
	downStart := strings.Index(content, "-- +goose Down")
	if upStart == -1 || downStart == -1 {
		t.Fatalf("migration %s missing +goose Up/Down markers", path)
	}
	return content[upStart+len("-- +goose Up") : downStart]
}

	// SET, and silently falls back to the default search_path.
	db.SetMaxOpenConns(1)

Confirmed the "18 integration tests" count is accurate. **Findings:** 1. **`DeleteExpired` has a TOCTOU race between the selection and the delete, on the exact predicate this batching logic exists to enforce.** The initial `SELECT id ... WHERE extraWhere AND clockColumn < $olderThan LIMIT $maxRows` correctly finds expired rows, but the per-row deletion is `DELETE FROM user_profile WHERE id = $1` — it never re-checks the expiry condition. If a profile is legitimately refreshed (e.g., IDP re-hydration bumps `updated_at` via `Upsert`) after being selected but before its turn in the delete loop, it gets deleted anyway even though it's no longer expired by the time the DELETE runs. On Postgres (READ COMMITTED, this engine's default), nothing prevents this — each statement gets its own snapshot, and the DELETE doesn't recheck. On CockroachDB it's likely self-healing (the concurrent write conflicts under SERIALIZABLE, forcing a `40001` retry that re-selects fresh candidates), so this is a real, silent bug specifically under the Postgres engine, not both. Fix: repeat the same `extraWhere AND clockColumn < $olderThan` predicate on the per-row `DELETE`, or fold the whole batch into a single `DELETE ... WHERE id = ANY($1) AND extraWhere AND clockColumn < $2` and use `RowsAffected` for the count instead of trusting the initial `SELECT`. 2. **Nil-vs-empty-slice on every multi-row read path** (`Search`, `Credentials().ListByUserID`, `Methods().List`): each declares `var results []T` and never initializes it, so "zero rows" returns a nil slice, not an empty one. `len(list) != 0` in the existing test passes either way and doesn't catch this. Functionally harmless in Go itself, but this DAO's output is headed straight for JSON serialization in the Q2 REST API — a nil slice marshals to `null`, an empty one to `[]`, and a client checking `response.results.length` against `null` will throw. Recommend `results := make([]model.T, 0, ...)` (or at minimum `[]model.T{}`) on all three paths now, before the handler layer inherits the inconsistency. 3. **Minor, not blocking:** `DeleteProfile` captures `source` from a `SELECT` before the `DELETE`, then writes it into `deletion_log`. If a caller updates a profile's `Source` between those two statements, the log would record a stale source. Narrow window, low consequence (an audit field, not the deletion itself), just noting it's the same TOCTOU shape as finding 1 in miniature. 4. **Also minor:** `OldestSurvivingAt`'s `SELECT MIN(clockColumn) WHERE extraWhere` rescans the entire remaining class on every batch call, not just the just-processed page — fine at this project's scale, but worth a one-line comment if a real deployment ever runs this over a multi-million-row class, since it turns an O(batches) sweep into O(batches × class size). Everything else — orphan-detection predicate (`EXISTS`/`NOT EXISTS` correctly scoped to `source = 'idp_cache'`, correctly computed at query time rather than stored, so a credential deletion correctly reclassifies a row without a stale-flag bug), error translation, retry-loop engine-gating, `Upsert`'s `created_at` exclusion, cascade behavior — checked out clean. 31	// violation message. SQLite's message for a named constraint (defined as
32	// `CONSTRAINT name CHECK (...)` in our own DDL) is deterministically
33	// "CHECK constraint failed: <name>", sometimes with a trailing " (<code>)"
34	// (observed from modernc.org/sqlite: "constraint failed: CHECK constraint
35	// failed: ck_user_credential_secret_state (275)") — this is not free-text
36	// message matching in the sense the contract prohibits; it's extracting a
37	// structured field SQLite's own error format guarantees to contain
38	// exactly the constraint name we ourselves wrote in DDL, the same
39	// artifact Postgres gives us through PgError.ConstraintName directly.
40	func checkConstraintName(msg string) (string, bool) {
41		const marker = "CHECK constraint failed: "
42		i := strings.LastIndex(msg, marker)
43		if i == -1 {
44			return "", false
45		}
46		name := strings.TrimSpace(msg[i+len(marker):])
47		// Strip a trailing " (<digits>)" result-code suffix, if present.
48		if j := strings.LastIndex(name, " ("); j != -1 && strings.HasSuffix(name, ")") {
